/**
 * EXTRACTION COMPLÈTE D'UN OU PLUSIEURS DRIVE → feuille "Export" (Chemin / Nom / Type / Lien)
 * Version PAR TRANCHES : pause à 5 min, reprise automatique là où on s'était arrêté.
 * >>> MODIFIÉ : multi-dossiers (ROOT_FOLDER_IDS) + reprise même après erreur
 *     + journal des dossiers inaccessibles (onglet _export_skips). <<<
 *
 * Format de sortie (inchangé, lu par Recherche_GEM_TV) :
 *   | Chemin | Nom | Type | Lien |   Type = "Dossier (racine)" | "Dossier" | "Fichier"
 *
 * MISE EN PLACE :
 *  1. https://script.google.com -> colle ce code -> Enregistre.
 *  2. testAcces()          -> vérifie l'accès à TOUS les dossiers racine.
 *  3. demarrerExport()     -> extraction NEUVE des deux drives (se relance seule).
 *  4. exportDrive()        -> REPREND un export interrompu (sans repartir de zéro).
 *  5. etatExport()         -> affiche où en est l'export (file restante, compteurs, skips).
 *  6. reinitialiserExport()-> remet tout à zéro.
 */

// Ajoute autant de dossiers racine que tu veux, séparés par des virgules.
const ROOT_FOLDER_IDS = [
  '1qDVkgIRe9wfkwm9f4sL1kRQQUNJXLzYq' // After-sales information
 
];
const SPREADSHEET_URL = ''; // laisse vide pour créer un classeur automatiquement
const OUTPUT_SHEET_NAME = 'Export';
const CSV_FILE_NAME = 'Recherche_GEM_TV_catalogue.csv';
const BUDGET_MS = 5 * 60 * 1000; // durée max d'une tranche (5 min)
const FLUSH_EVERY = 1500;        // écrit dans la feuille tous les N lignes
const ENVOYER_EMAIL = true;
const EMAIL_DESTINATAIRE = 'jerome.leconte@fnacdarty.com';
// =============================

const FEUILLE_QUEUE = '_export_queue';
const FEUILLE_SKIPS = '_export_skips';

/**
 * Traite UNE tranche (≤ 5 min) puis se relance si besoin.
 * File : {t:'F',id,path} ouvrir dossier · {t:'SF',tok,path} sous-dossiers · {t:'FF',tok,path} fichiers
 */
function exportDrive() {
  const DEBUT = Date.now();
  const props = PropertiesService.getScriptProperties();
  supprimerTriggers_('_continuerExport');

  const ss = getSpreadsheet_();
  const shOut = getOutputSheet_(ss);
  const shQueue = getFeuilleCachee_(ss, FEUILLE_QUEUE);
  const shSkip = getFeuilleCachee_(ss, FEUILLE_SKIPS);

  try {
    // Démarrage d'une nouvelle extraction si aucune n'est en cours
    const enCours = props.getProperty('export_enCours') === 'true';
    if (!enCours) {
      shOut.clearContents();
      shOut.getRange(1, 1, 1, 4).setValues([['Chemin', 'Nom', 'Type', 'Lien']]);
      shOut.setFrozenRows(1);
      shSkip.clearContents();
      shSkip.getRange(1, 1, 1, 2).setValues([['Chemin sauté', 'Raison']]);

      // Une racine + une entrée de file PAR dossier de ROOT_FOLDER_IDS.
      const filesInitiale = [];
      let ligne = 2;
      ROOT_FOLDER_IDS.forEach(function (id) {
        id = String(id).trim();
        if (!id) return;
        let root;
        try { root = DriveApp.getFolderById(id); }
        catch (e) { journaliserSkip_(shSkip, '(racine) ' + id, e.message); return; }
        const rootName = root.getName();
        shOut.getRange(ligne++, 1, 1, 4).setValues([['', rootName, 'Dossier (racine)', root.getUrl()]]);
        filesInitiale.push({ t: 'F', id: id, path: rootName });
      });
      ecrireQueue_(shQueue, filesInitiale);

      props.setProperty('export_enCours', 'true');
      props.setProperty('export_nbFichiers', '0');
      props.setProperty('export_nbDossiers', '0');
    }

    const queue = lireQueue_(shQueue);
    let buffer = [];
    let nbF = 0, nbD = 0;

    const budgetDepasse = function () { return (Date.now() - DEBUT) >= BUDGET_MS; };

    while (queue.length > 0 && !budgetDepasse()) {
      const item = queue.shift();

      if (item.t === 'F') {
        let folder;
        try { folder = DriveApp.getFolderById(item.id); }
        catch (e) { journaliserSkip_(shSkip, item.path, 'Dossier inaccessible : ' + e.message); continue; }
        queue.push({ t: 'SF', tok: folder.getFolders().getContinuationToken(), path: item.path });
        queue.push({ t: 'FF', tok: folder.getFiles().getContinuationToken(), path: item.path });

      } else if (item.t === 'SF') {
        let it;
        try { it = DriveApp.continueFolderIterator(item.tok); }
        catch (e) { journaliserSkip_(shSkip, item.path, 'Jeton sous-dossier invalide : ' + e.message); continue; }
        while (it.hasNext()) {
          if (budgetDepasse()) { queue.unshift({ t: 'SF', tok: it.getContinuationToken(), path: item.path }); break; }
          const sf = it.next();
          buffer.push([item.path, sf.getName(), 'Dossier', sf.getUrl()]);
          nbD++;
          queue.push({ t: 'F', id: sf.getId(), path: item.path + '/' + sf.getName() });
        }

      } else if (item.t === 'FF') {
        let it;
        try { it = DriveApp.continueFileIterator(item.tok); }
        catch (e) { journaliserSkip_(shSkip, item.path, 'Jeton fichier invalide : ' + e.message); continue; }
        while (it.hasNext()) {
          if (budgetDepasse()) { queue.unshift({ t: 'FF', tok: it.getContinuationToken(), path: item.path }); break; }
          const f = it.next();
          buffer.push([item.path, f.getName(), 'Fichier', f.getUrl()]);
          nbF++;
        }
      }

      if (buffer.length >= FLUSH_EVERY) { flush_(shOut, buffer); buffer = []; }
    }

    if (buffer.length) flush_(shOut, buffer);
    ecrireQueue_(shQueue, queue);

    nbF += parseInt(props.getProperty('export_nbFichiers') || '0', 10);
    nbD += parseInt(props.getProperty('export_nbDossiers') || '0', 10);
    props.setProperty('export_nbFichiers', String(nbF));
    props.setProperty('export_nbDossiers', String(nbD));

    if (queue.length > 0) {
      planifierContinuation_();
      Logger.log('Tranche OK. Reste ' + queue.length + ' en file. Cumul : '
        + nbD + ' dossier(s), ' + nbF + ' fichier(s). Reprise auto dans 1 min.');
      return;
    }

    // Terminé
    props.setProperty('export_enCours', 'false');
    props.deleteProperty('export_nbFichiers');
    props.deleteProperty('export_nbDossiers');
    const nbSkips = Math.max(0, shSkip.getLastRow() - 1);
    Logger.log('✅ TERMINÉ : ' + nbD + ' dossier(s), ' + nbF + ' fichier(s), '
      + nbSkips + ' dossier(s) sauté(s). Classeur : ' + ss.getUrl());
    if (ENVOYER_EMAIL) {
      MailApp.sendEmail({
        to: EMAIL_DESTINATAIRE,
        subject: 'Extraction Drive terminée : ' + nbD + ' dossiers, ' + nbF + ' fichiers',
        htmlBody: '<p>Extraction terminée.</p><p>' + nbD + ' dossier(s) · ' + nbF + ' fichier(s) · '
          + nbSkips + ' dossier(s) sauté(s) (voir onglet "' + FEUILLE_SKIPS + '").</p>'
          + '<p>Classeur : <a href="' + ss.getUrl() + '">ouvrir</a>.</p>'
      });
    }

  } catch (err) {
    // La chaîne ne doit JAMAIS casser : on reprogramme une reprise même en cas d'erreur.
    Logger.log('⚠️ Erreur pendant la tranche : ' + err.message + ' — reprise programmée dans 2 min.');
    ScriptApp.newTrigger('_continuerExport').timeBased().after(2 * 60 * 1000).create();
  }
}

/** Relancée par le déclencheur de reprise (ne pas exécuter à la main). */
function _continuerExport() { exportDrive(); }

/** Extraction NEUVE (repart de zéro) puis enchaîne les tranches. */
function demarrerExport() {
  reinitialiserExport();
  exportDrive();
}

/* ===================== ÉCRITURE ===================== */

function flush_(shOut, buffer) {
  if (!buffer.length) return;
  shOut.getRange(shOut.getLastRow() + 1, 1, buffer.length, 4).setValues(buffer);
}

function journaliserSkip_(shSkip, chemin, raison) {
  Logger.log('SKIP : ' + chemin + ' — ' + raison);
  shSkip.getRange(shSkip.getLastRow() + 1, 1, 1, 2).setValues([[chemin, raison]]);
}

/* ===================== SPREADSHEET / FEUILLES ===================== */

function getSpreadsheet_() {
  if (SPREADSHEET_URL) return SpreadsheetApp.openByUrl(SPREADSHEET_URL);
  const props = PropertiesService.getScriptProperties();
  const id = props.getProperty('export_ss_id');
  if (id) { try { return SpreadsheetApp.openById(id); } catch (e) { /* recréé ci-dessous */ } }
  const ss = SpreadsheetApp.create('Export_GEM_TV_Drive');
  props.setProperty('export_ss_id', ss.getId());
  Logger.log('Nouveau classeur créé : ' + ss.getUrl());
  return ss;
}

function getOutputSheet_(ss) {
  let sh = ss.getSheetByName(OUTPUT_SHEET_NAME);
  if (!sh) sh = ss.insertSheet(OUTPUT_SHEET_NAME);
  return sh;
}

function getFeuilleCachee_(ss, nom) {
  let sh = ss.getSheetByName(nom);
  if (!sh) { sh = ss.insertSheet(nom); sh.hideSheet(); }
  return sh;
}

/* ===================== FILE D'ATTENTE ===================== */

function lireQueue_(sh) {
  const last = sh.getLastRow();
  if (last < 1) return [];
  const vals = sh.getRange(1, 1, last, 1).getValues();
  const out = [];
  for (let i = 0; i < vals.length; i++) {
    if (vals[i][0]) { try { out.push(JSON.parse(vals[i][0])); } catch (e) { /* ignore */ } }
  }
  return out;
}

function ecrireQueue_(sh, arr) {
  sh.clearContents();
  if (!arr.length) return;
  const data = arr.map(function (x) { return [JSON.stringify(x)]; });
  sh.getRange(1, 1, data.length, 1).setValues(data);
}

/* ===================== DÉCLENCHEURS ===================== */

function planifierContinuation_() {
  supprimerTriggers_('_continuerExport');
  ScriptApp.newTrigger('_continuerExport').timeBased().after(60 * 1000).create();
}

function supprimerTriggers_(handler) {
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (t.getHandlerFunction() === handler) ScriptApp.deleteTrigger(t);
  });
}

/* ===================== OUTILS À LANCER À LA MAIN ===================== */

/** Remet l'extraction à zéro (état, file, compteurs). */
function reinitialiserExport() {
  const props = PropertiesService.getScriptProperties();
  ['export_enCours', 'export_nbFichiers', 'export_nbDossiers'].forEach(function (k) { props.deleteProperty(k); });
  supprimerTriggers_('_continuerExport');
  const ss = getSpreadsheet_();
  [FEUILLE_QUEUE, FEUILLE_SKIPS].forEach(function (n) {
    const q = ss.getSheetByName(n); if (q) ss.deleteSheet(q);
  });
  Logger.log('Extraction réinitialisée.');
}

/** Où en est l'export : file restante, compteurs, nb de dossiers sautés. */
function etatExport() {
  const props = PropertiesService.getScriptProperties();
  const ss = getSpreadsheet_();
  const q = ss.getSheetByName(FEUILLE_QUEUE);
  const s = ss.getSheetByName(FEUILLE_SKIPS);
  const reste = q ? q.getLastRow() : 0;
  const skips = s ? Math.max(0, s.getLastRow() - 1) : 0;
  const enCours = props.getProperty('export_enCours');
  Logger.log('enCours=' + enCours + ' · file restante=' + reste
    + ' · dossiers=' + (props.getProperty('export_nbDossiers') || '?')
    + ' · fichiers=' + (props.getProperty('export_nbFichiers') || '?')
    + ' · dossiers sautés=' + skips
    + (reste > 0 ? '  -> lance exportDrive() pour reprendre.' : '  -> terminé.'));
}

/** Test d'accès à TOUS les dossiers racine + au classeur. */
function testAcces() {
  ROOT_FOLDER_IDS.forEach(function (id) {
    id = String(id).trim();
    if (!id) return;
    try {
      const root = DriveApp.getFolderById(id);
      Logger.log('Racine OK : ' + root.getName() + ' (' + id + ')');
    } catch (e) { Logger.log('ERREUR racine ' + id + ' : ' + e.message); }
  });
  try {
    const ss = getSpreadsheet_();
    Logger.log('Classeur OK : ' + ss.getName() + ' -> ' + ss.getUrl());
  } catch (e) { Logger.log('ERREUR classeur : ' + e.message); }
}

/** (Option) Génère/actualise un CSV dans le Drive à partir de la feuille "Export". */
function exporterCsvDansDrive() {
  const ss = getSpreadsheet_();
  const sh = ss.getSheetByName(OUTPUT_SHEET_NAME);
  if (!sh) { Logger.log('Feuille "' + OUTPUT_SHEET_NAME + '" introuvable.'); return; }
  const values = sh.getDataRange().getValues();
  const csv = values.map(function (row) {
    return row.map(function (c) {
      let s = (c === null || c === undefined) ? '' : String(c);
      if (/[",\n\r]/.test(s)) s = '"' + s.replace(/"/g, '""') + '"';
      return s;
    }).join(',');
  }).join('\r\n');
  const existing = DriveApp.getFilesByName(CSV_FILE_NAME);
  let file;
  if (existing.hasNext()) { file = existing.next(); file.setContent('﻿' + csv); }
  else { file = DriveApp.createFile(Utilities.newBlob('﻿' + csv, 'text/csv', CSV_FILE_NAME)); }
  Logger.log('CSV écrit : ' + file.getUrl());
  return file.getUrl();
}
